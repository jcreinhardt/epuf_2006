\section{A new method to estimate earnings cross-sections}
% ======================================================================

% ======================================================================
% Slide 1 --- Old vs. new
% ======================================================================
\begin{frame}{Old vs.\ new cross-sectional estimation}
  \textbf{Old:} \textbf{GMM} on moments collected from \textbf{several different sources}.
  Each source has its own sample, so the moments being matched do not come from one population.

  \medskip
  \textbf{New:} \textbf{MLE} on the \textbf{SSA microdata alone} --- the sample that actually
  pays into Social Security:
  \begin{enumerate}
    \item Fit \textbf{parametric distributions} per $(\text{year},\text{sex},\text{age})$ cell
          on the \textbf{1951--2006 SSA public-use file} (EPUF).
    \item \textbf{Extrapolate} the parameter surfaces to the non-covered years:
          backward to \textbf{1937--1950}, forward to \textbf{2006--2100}.
    \item Matches the \textbf{ASS aggregate totals} and the \textbf{distribution of earnings}
          better than the GMM fit.
  \end{enumerate}
  \vfill
  \begin{center}
  \begin{tikzpicture}[node distance=6mm,
      every node/.style={font=\footnotesize,align=center,draw,rounded corners,
                         inner sep=4pt,draw=mDarkTeal!40}]
    \node[fill=mDarkTeal!5] (a) {1937--1950\\\scriptsize extrapolated};
    \node[fill=mLightBrown!25,right=of a] (b) {1951--2006\\\scriptsize fitted};
    \node[fill=mDarkTeal!5,right=of b] (c) {2006--2100\\\scriptsize extrapolated};
    \node[fill=white,right=10mm of c] (d) {+ dynamics\\\scriptsize ordinal transform};

  \end{tikzpicture}
  \end{center}
\end{frame}

% Slide 3 --- Step 1, the estimator
% ======================================================================
\begin{frame}{Step 1 --- the estimator}
  \small
  {\footnotesize Cell $c=(\text{year }k,\text{ sex }s,\text{ age }a)$, $y_i=\log x_i$, censoring
  at $\LOWC=\$200$, $\HIGHC=\text{taxmax}_k-\$1000$. $f_\theta$: dPLN (men,
  $\theta=(\alpha,\beta,\nu,\tau)$), log-normal mixture (women,
  $\theta=(\mu_1,\mu_2,\sigma_1,\sigma_2,w)$).\par}

  \smallskip
  \textbf{1. Doubly censored log-likelihood} --- outside $[\LOWC,\HIGHC]$ only the counts
  $n^{\text{lo}}_c,n^{\text{hi}}_c$ are observed:
  \vspace{-5pt}
  \[
    \ell_c(\theta)=\!\!\sum_{i:\,\LOWC<x_i<\HIGHC}\!\!\log f_\theta(y_i)
      \ +\ n^{\text{lo}}_c\log F_\theta(\log\LOWC)
      \ +\ n^{\text{hi}}_c\log\bigl[1-F_\theta(\log\HIGHC)\bigr]
  \]
  \vspace{-13pt}

  \textbf{2. $\alpha$ not identified off the interior} --- up to \textbf{80\%} of a cell's
  observations are censored:
  \vspace{-5pt}
  \[
    \operatorname*{arg\,max}_{\{\theta_c\},\,\eta_k}\ \sum_{c\in k}\ell_c(\theta_c)
      -\eta_k\Bigl(\sum_{c\in k}w_c\,\mathbb{E}_{\theta_c}[X]-m_k\Bigr)
    \;=\;
    \operatorname*{arg\,max}_{\eta_k}\ \max_{\{\theta_c\}}\
      \sum_{c\in k}\Bigl[\ell_c(\theta_c)-\eta_k\,w_c\,\mathbb{E}_{\theta_c}[X]\Bigr]
  \]
  \vspace{-13pt}

  \textbf{3. Regime switching} --- some ages have \textbf{two minima far apart in $\theta$ with
  very similar likelihoods}. Penalize \textbf{functionals $g$ of the distribution} instead of
  raw parameters:
  \vspace{-4pt}
  \[
    \min_{\{\theta_c\}}\ \sum_{c\in k}\Bigl[-\ell_c(\theta_c)
      +\eta_k\,w_c\,\mathbb{E}_{\theta_c}[X]\Bigr]
      \;+\;\rho\sum_{s}\bigl\|D\,G_s\,\Omega^{1/2}\bigr\|_F^2
  \]
  \vspace{-11pt}

  {\footnotesize $w_c=n_c/n_k$ (ages 15--77), $m_k=$ ASS \texttt{aggearn\_tot/num\_wrk};
  rows of $G_s$: $g(\theta_{s,a})^{\top}$ by age; $D$ = second-difference matrix along age;
  $\Omega^{1/2}=\operatorname{diag}\sqrt{\Omega_j}$.\par}
\end{frame}

% ======================================================================
% Slide 4 --- how it is solved
% ======================================================================
\begin{frame}{Step 1 --- how it is solved}
  Fix $\rho$. Per year:
  \begin{itemize}
    \item[\textbf{0.}] Unconstrained MLE per cell $\Rightarrow$ starting values and the weight
      matrix $\Omega_j=1/\operatorname{MAD}_a[(\Delta^2 g)_j]^2$.
    \item[\textbf{1.}] Find $\eta_k$ such that the constraint holds.
  \end{itemize}
  \vfill
  Then compare across $\rho$ and pick the one where model and ASS totals overlap best.
\end{frame}

% ======================================================================
% Slide 5 --- Steps 2 and 3
% ======================================================================
\begin{frame}{Steps 2 \& 3 --- extrapolate, then aggregate}
  \small
  \textbf{Step 2 --- extrapolate to 1937--1950 and 2005--2100.} Per $(\text{sex},\text{age})$:
  \begin{itemize}\itemsep1pt
    \item Shapes --- $(\alpha,\beta,\tau)$ men, $(\sigma_1,\sigma_2,w)$ women --- held fixed at
      the \textbf{2000--04 average} forward, the \textbf{1951--55 average} backward.
    \item Locations --- $\nu$ men, $(\mu_1,\mu_2)$ women --- shifted by published earnings growth:
      \vspace{-3pt}
      \[
        m_{s,a}(y)=m_{s,a}^{0}+\bigl[G(y)-\bar G\bigr]
      \]
      \vspace{-13pt}

      {\footnotesize $G$ = log average covered earnings: ASS realized through 2004, Trustees
      Report projected growth after.\par}
    \item 1937--1950 only: men's $\alpha$ additionally scaled by $k_y$, solved per year so the
      uncapped mean matches ASS exactly ($k_y\alpha>1$ imposed).
  \end{itemize}

  \smallskip
  \textbf{Step 3 --- aggregate} to yearly taxable-earnings totals, using employment shares and
  labor-force totals from ASS / TR. Validated against ASS + TR, capped and uncapped.
\end{frame}

\section{What did CMS do?}
% ======================================================================

% ======================================================================
% Slide 9 --- CMS overview
% ======================================================================
\begin{frame}{Log-normal correction}
  \small
  \begin{enumerate}\itemsep2pt
    \item Keep the \textbf{GKOS (2021)} process --- mixture AR(1) $z$, mixture transitory
      $\varepsilon$, HIP $(\alpha_i,\beta_i)$, nonemployment --- estimated on \emph{male} SSA
      data \textbf{jointly with a latent} $g(t)$.
    \item Replace $g(t)$ by $\tilde g_{c,s}(t)$: a cubic in age fit to \textbf{GKSW (2017)
      cohort $\times$ gender profiles of \emph{mean log} earnings}.
    \item Pick $\tilde g$ so that means line up:
      \vspace{-2pt}
      \[
        \mathbb{E}[\tilde Y\mid t,c,s]
          =\exp\bigl(\underbrace{\mathbb{E}[\log Y\mid t,c,s]}_{\text{observed}}\bigr)
      \]
      \vspace{-4pt}
      \[
        \tilde Y_{it}=\exp\bigl(\underbrace{\tilde g_{c,s}(t)}_{\text{deterministic}}
          +\underbrace{\alpha_i+\beta_i t+z_{it}+\varepsilon_{it}}_{x_{it}\ \approx\ N(0,V)}\bigr),
        \qquad
        \mathbb{E}[\tilde Y]=e^{\tilde g}\,\mathbb{E}[e^{x}]=e^{\tilde g+V/2}
      \]
      \vspace{-2pt}
      \[
        \Longrightarrow\quad
        \tilde g_{c,s}(t)=\mathbb{E}[\log Y\mid t,c,s]-\tfrac12 V
      \]
  \end{enumerate}
\end{frame}

% ======================================================================
% Slide 10 --- the nonemployment fix
% ======================================================================
\begin{frame}{CMS: the nonemployment fix}
  \small
  \textbf{Problem:} non-claiming rates in the SCF are far higher than the process implies.

  \medskip
  \textbf{Fix:} multiply earnings by a nonemployment dummy $u_{i}\in\{0,1\}$, fixed at
  \textbf{10\%} (men) / \textbf{20\%} (women). That moves the mean:
  \vspace{-2pt}
  \[
    \mathbb{E}\bigl[(1-u)\tilde Y\bigr]
      =\mathbb{E}[\tilde Y]\,\mathbb{E}[1-u]+\operatorname{Cov}(\tilde Y,1-u)
  \]
  \vspace{-15pt}

  \medskip
  $u$ is drawn \textbf{independently} of $\tilde Y$, so $\operatorname{Cov}(\tilde Y,1-u)=0$ and
  dividing by $\mathbb{E}[1-u]$ restores the mean \textbf{exactly}:
  \vspace{-2pt}
  \[
    \tilde Y_{it}=\frac{(1-u_{i})\,e^{\tilde g_{c,s}(t)+x_{it}}}{\mathbb{E}[1-u]}
    \qquad\Longrightarrow\qquad
    \mathbb{E}[\tilde Y_{it}]=e^{\tilde g_{c,s}(t)+V/2}
  \]
\end{frame}

\begin{frame}{Problem --- selection bias}
  \small
  The realized process is $(1-\nu_{it})\exp(\tilde g_{c,s}(t)+x_{it})$, and nonemployment
  depends on the persistent component:
  $\ \Pr(\nu_{it}>0)=\Lambda(a+b\,t+\alert{c\,z_{it}}+d\,z_{it}t)$, $c=-5.03$. It hits
  \textbf{low-$z$} workers, so $\mathbb{E}[\log Y\mid\text{observed}]\neq\tilde g_{c,s}(t)$.

\section{CMS vs.\ ordinal transform}
% ======================================================================

% ======================================================================
% Slide --- the mean does not matter under an ordinal transform
% ======================================================================
\begin{frame}{After an ordinal transform, the mean does not matter}
  \small
  $(X,Y)$ with marginals $F_X,F_Y$ and copula $C$; target marginals $\tilde F_X,\tilde F_Y$:
  \vspace{-3pt}
  \[
    (\tilde X,\tilde Y)=\bigl(\tilde F_X^{-1}(F_X(X)),\ \tilde F_Y^{-1}(F_Y(Y))\bigr)
    \qquad\Longrightarrow\qquad
    \Pr(\tilde X\le x,\ \tilde Y\le y)=C\bigl(\tilde F_X(x),\tilde F_Y(y)\bigr)
  \]
  \vspace{-12pt}

  Shift the inputs --- by Sklar's theorem,
  \vspace{-3pt}
  \[
    \Pr(X+a\le x,\ Y+b\le y)=C\bigl(F_X(x-a),\ F_Y(y-b)\bigr)
  \]
  \vspace{-12pt}
  \begin{itemize}\itemsep2pt
    \item Shifts enter the \textbf{marginals only}; $C$ is invariant under increasing maps.
    \item The transform replaces exactly those marginals with $\tilde F_X,\tilde F_Y$
      $\Rightarrow$ same $C\bigl(\tilde F_X(x),\tilde F_Y(y)\bigr)$ with or without the shift.
    \item $\tilde g_{c,s}(t)$, $-\tfrac12V$, $1/\mathbb{E}[1-u]$ are within-cell shifts in logs
      $\Rightarrow$ \textbf{CMS and GKOS give the exact same transformed process}.
  \end{itemize}
\end{frame}
